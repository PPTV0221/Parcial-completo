# El repositorio ya trae commits hechos. Para subirlo a GitHub:
# 1) Crea un repositorio vacío en https://github.com/new (nombre: PawCare, sin README)
# 2) Ejecuta:
git remote add origin https://github.com/TU_USUARIO/PawCare.git
git push -u origin main

# (Opcional) poner tu nombre/correo en los commits ya creados:
git config user.name  "Tu Nombre"
git config user.email "tu@correo.com"
git rebase -r --root --exec "git commit --amend --reset-author --no-edit"
